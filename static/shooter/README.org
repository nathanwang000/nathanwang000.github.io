// Helper function to get a random visual
function getRandomVisual(mediaType, category) {
    if (!visualLibraries[mediaType] || !visualLibraries[mediaType][category]) {
        return null;
    }

    const collection = visualLibraries[mediaType][category];
    const item = collection[Math.floor(Math.random() * collection.length)];

    return {
        type: mediaType,
        content: item,
        size: 0.04 * canvas.height // Default size
    };
}
